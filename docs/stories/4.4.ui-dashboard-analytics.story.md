## Story

**As a** usuário,
**I want** ver meu analytics em `/dashboard/analytics` com 4 cards de totais (Total Page Views/Clicks lifetime + Page Views/Clicks 30d), gráfico de linha 7d/30d (toggle via URL `?range=`), tabela de cliques por link (title, url, clicks 7d, clicks 30d, total) ordenada desc, empty state ilustrado quando sem eventos, loading skeletons durante fetch, e acessibilidade completa (chart com fallback de tabela + headers semânticos),
**so that** eu entenda quais links geram engajamento e tome decisões data-driven sobre meu BioLink — fechando, junto com o backbone SQL 4.1/4.2/4.3, o ciclo analítico do MVP (FR11) sem violar NFR2 (Lighthouse ≥ 90) nem NFR13 (browser support iOS 14+).

    ```typescript
    function gapFill(series: { day: string; count: number }[], days: 7 | 30) {
      const today = new Date();
      const result: { day: string; count: number }[] = [];
      for (let i = days - 1; i >= 0; i--) {
        const d = new Date(today);
        d.setDate(d.getDate() - i);
        const isoDay = d.toISOString().slice(0, 10); // YYYY-MM-DD
        const match = series.find((s) => s.day === isoDay);
        result.push({ day: isoDay, count: match?.count ?? 0 });
      }
      return result;
    }
    ```

### Queries SQL canônicas (Task 2.4-2.6 — fonte da verdade)

```sql
-- AC1: Total Page Views (lifetime) — count head request
GET /rest/v1/page_views?select=id&page_id=eq.<alicePageId>
   Headers: Prefer: count=exact, Range-Unit: items, Range: 0-0
   Response: 204 + Content-Range: 0-0/<N>
-- (Supabase JS: .from('page_views').select('id', { count: 'exact', head: true }).eq('page_id', pageId))

-- AC1: Total Clicks (lifetime) — needs linkIds first
GET /rest/v1/links?select=id&page_id=eq.<alicePageId>
   → [{ id: <linkId1> }, { id: <linkId2> }, ...]

GET /rest/v1/click_events?select=id&link_id=in.(<linkId1>,<linkId2>,...)
   Headers: Prefer: count=exact, head=true
-- (Supabase JS: .from('click_events').select('id', { count: 'exact', head: true }).in('link_id', linkIds))

-- AC1: Page Views (últimos 30d) — view + sum client-side
GET /rest/v1/page_views_30d?select=count&page_id=eq.<alicePageId>
   → [{ count: 4 }, { count: 2 }, { count: 1 }]  -- sparse rows por dia
-- Sum: 4 + 2 + 1 = 7

-- AC1: Clicks (últimos 30d) — view filtered by linkIds
GET /rest/v1/link_clicks_30d?select=count&link_id=in.(<linkIds>)
   → [{ count: N }, ...]
-- Sum: client-side

-- AC2: Série temporal 7d ou 30d (RPC)
POST /rest/v1/rpc/get_page_views_series
   Body: { "p_page_id": "<alicePageId>", "p_days": 7 }
   → [{ day: '2026-05-25', count: 4 }, { day: '2026-05-27', count: 5 }, ...]  -- sparse, ASC

-- AC3: Tabela por link — links + 3 aggregations
GET /rest/v1/links?select=id,title,url&page_id=eq.<alicePageId>&order=position.asc
GET /rest/v1/link_clicks_7d?select=link_id,count&link_id=in.(<linkIds>)
GET /rest/v1/link_clicks_30d?select=link_id,count&link_id=in.(<linkIds>)
GET /rest/v1/click_events?select=link_id&link_id=in.(<linkIds>)  -- group by client-side
-- Merge + sort desc by clicksTotal in page.tsx
```
