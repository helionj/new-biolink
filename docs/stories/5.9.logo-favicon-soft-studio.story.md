   ```tsx
   import * as React from 'react';
   import Link from 'next/link';
   import { cn } from '@/lib/utils';

   const SIZE_CLASSES = {
     sm: 'text-base', // 16px per spec §1.6 L351 "16px header"
     md: 'text-2xl', // 24px per spec §1.6 L351 "24px sidebar dashboard"
     lg: 'text-[2.5rem]', // 40px per spec §1.6 L351 "40px footer landing"
   };

   type WordmarkProps = {
     size?: 'sm' | 'md' | 'lg';
     showStar?: boolean;
     className?: string;
     href?: string;
   };

   /**
    * Wordmark primitive — Soft Studio Phase 5 (Story 5.9).
    * Canonical brand mark "biolink ★" consumed across landing/auth/dashboard headers.
    * Spec §1.6 L347-354 verbatim:
    *   - "biolink" all-lowercase (Q5 §6 ratified)
    *   - DM Sans Bold (700), tracking -0.04em
    *   - Color: var(--primary) (deep plum light/brand, lavender dark)
    *   - Sizes: sm (16) / md (24) / lg (40)
    *   - ★ asterisco placeholder (Q4 §6 ratified; logomark real Phase 2)
    */
   function Wordmark({ size = 'md', showStar = true, className, href }: WordmarkProps) {
     const content = (
       <span
         className={cn(
           'inline-flex items-baseline gap-1 font-bold tracking-[-0.04em] text-primary',
           SIZE_CLASSES[size],
           className,
         )}
       >
         <span>biolink</span>
         {showStar && (
           <span aria-hidden="true" className="text-accent">
             ★
           </span>
         )}
       </span>
     );
     if (href) {
       return (
         <Link href={href} aria-label="biolink" className="inline-flex">
           {content}
         </Link>
       );
     }
     return content;
   }

   export { Wordmark };
   ```

   Specs: (a) `font-bold` = 700 weight per spec L349 "DM Sans Bold (700)"; (b) `tracking-[-0.04em]` arbitrary value per spec L349 verbatim; (c) `text-primary` consume token `--primary` plum light/brand + lavender dark per spec L350; (d) ★ asterisco em `text-accent` (peach `var(--accent)` per spec §1.6 L354 favicon hint OR plum primary — spec L344 menciona "asterisco como pivô final futuro logomark hook" sem cor explicit; usar `text-accent` peach para visual differentiation do "biolink" plum — pop visual coerente com landing/dashboard palette); (e) `aria-hidden="true"` no ★ (decoração visual; `aria-label="biolink"` no `<Link>` carrega screen reader announcement); (f) `aria-label` em Link container = "biolink" pure (ignora asterisco para SR clarity); (g) `inline-flex` ao invés de `<a className="inline-flex">` permite Wordmark consumido as plain `<span>` em footer/headings/etc. via opt-out `href` prop; (h) className passthrough via `cn()` + tailwind-merge.
