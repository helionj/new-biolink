    ```tsx
    'use client';

    import { useRef, useState, useTransition } from 'react';
    import { useRouter } from 'next/navigation';
    import Image from 'next/image';

    import { Avatar, AvatarFallback, AvatarImage } from '@/components/ui/avatar';
    import { Button } from '@/components/ui/button';
    import { toast } from '@/lib/toast';
    import { uploadAvatar } from '@/server/profile/actions';

    type Props = {
      currentAvatarUrl: string | null;
      displayName: string | null;
      username: string;
    };

    export function AvatarUpload({ currentAvatarUrl, displayName, username }: Props) {
      const router = useRouter();
      const inputRef = useRef<HTMLInputElement>(null);
      const [isPending, startTransition] = useTransition();
      const [previewUrl, setPreviewUrl] = useState<string | null>(currentAvatarUrl);

      function handleFileChange(e: React.ChangeEvent<HTMLInputElement>) {
        const file = e.target.files?.[0];
        if (!file) return;

        // Preview otimista — revogado em rollback (DEV-2 — optimistic UI verbatim Story 3.3).
        const localUrl = URL.createObjectURL(file);
        setPreviewUrl(localUrl);

        const formData = new FormData();
        formData.append('avatar', file);

        startTransition(async () => {
          const res = await uploadAvatar(formData);
          URL.revokeObjectURL(localUrl); // cleanup memory leak
          if (!res.ok) {
            setPreviewUrl(currentAvatarUrl); // rollback
            toast.error(res.error);
            return;
          }
          setPreviewUrl(res.data.avatar_url);
          toast.success('Avatar atualizado');
          router.refresh(); // RSC re-fetch profile no parent
        });
      }

      const initials = (displayName ?? username).slice(0, 2).toUpperCase();

      return (
        <div className="flex items-center gap-4">
          <Avatar size="lg">
            {previewUrl ? (
              <AvatarImage
                render={
                  <Image
                    src={previewUrl}
                    alt={`Avatar de ${displayName ?? username}`}
                    width={40}
                    height={40}
                  />
                }
              />
            ) : (
              <AvatarFallback>{initials}</AvatarFallback>
            )}
          </Avatar>
          <div className="space-y-1">
            <input
              ref={inputRef}
              type="file"
              accept="image/jpeg,image/png,image/webp"
              className="sr-only"
              onChange={handleFileChange}
              disabled={isPending}
            />
            <Button
              type="button"
              variant="outline"
              size="sm"
              onClick={() => inputRef.current?.click()}
              disabled={isPending}
            >
              {isPending ? 'Enviando…' : 'Trocar avatar'}
            </Button>
            <p className="text-xs text-muted-foreground">jpg, png ou webp — máx 1 MB</p>
          </div>
        </div>
      );
    }
    ```

### Decisão DEV-2 — Optimistic preview com `URL.createObjectURL` + rollback

**Problema:** o user clica "Trocar avatar" e espera feedback visual imediato — `<Image>` re-renderizado para o novo arquivo antes do upload finalizar. Padrões:

| Padrão                                   | Latência percebida | Complexidade |
| ---------------------------------------- | ------------------ | ------------ |
| Pessimista (esperar upload)              | ~1-3s (network)    | Baixa        |
| **Optimistic via `URL.createObjectURL`** | **~0ms**           | **Média**    |
| Base64 inline preview                    | ~0ms               | Média        |

**Resolução:** `URL.createObjectURL` + `useState` + `useTransition` (padrão verbatim Story 3.3 DEV-4 — `<ThemeSelector>` usa o mesmo). `URL.revokeObjectURL` no cleanup (evita memory leak — caveat do Web API). Em erro do Server Action, reverter `previewUrl` para `currentAvatarUrl` (rollback) e mostrar `toast.error`.

### Decisão DEV-3 — `getPublicUrl` vs. `createSignedUrl`

**Problema:** o objeto em `storage.objects` pode ser acessado via:

| Método            | Latência     | Bandwidth | Segurança                             |
| ----------------- | ------------ | --------- | ------------------------------------- |
| `getPublicUrl`    | 0 (cliente)  | direto    | RLS `avatars_select_public` (público) |
| `createSignedUrl` | 1 round-trip | overhead  | URL temporária assinada (1h default)  |

**Resolução:** **`getPublicUrl`** — alinhado com `bucket public: true` (Task 1.2) + path quente da página pública SSR (`<PublicPage>` Story 2.7) que renderiza o avatar via `next/image` priority. Avatar é conteúdo público por design (FR13 — "página pública mostra avatar"); signed URL seria over-engineering. Trade-off: URL persistida em `profiles.avatar_url` não expira (oposto de signed URL); rotação ao trocar avatar resolve.

### Decisão DEV-4 — Path determinístico com overwrite

**Problema:** ao trocar avatar, o objeto anterior fica órfão se o novo upload usar nome diferente.

**Opções:**

| Pattern                               | Objetos órfãos             | URL `avatar_url` precisa update?                |
| ------------------------------------- | -------------------------- | ----------------------------------------------- |
| `{user.id}/{timestamp}.jpg`           | Sim (acumula a cada troca) | Sim                                             |
| `{user.id}/{uuid}.jpg`                | Sim                        | Sim                                             |
| **`{user.id}/avatar.{ext}` + upsert** | **Não (overwrite)**        | **Não (path estável; URL muda só se ext muda)** |

**Resolução:** path **determinístico** `{user.id}/avatar.{ext}` + `upload(..., { upsert: true })`. Vantagens:

1. Sem objetos órfãos no bucket (zero cleanup necessário).
2. URL em `profiles.avatar_url` muda apenas se a extensão mudar (jpg→png). Cache busting via query string opcional (`?v=<updated_at>`) se necessário.
3. RLS `avatars_update_own` (Task 1.3) cobre o overwrite verbatim (mesmo `WITH CHECK` que `avatars_insert_own`).

Trade-off: trocar jpg→png deixa o `.jpg` antigo órfão (extensão diferente = path diferente). Mitigação: best-effort `supabase.storage.from('avatars').remove([oldPath])` em Server Action quando detectar ext change. **Decisão pragmática (DEV-4):** **não** implementar cleanup de extensão antiga no MVP — overhead de query do `profiles.avatar_url` atual para parsear ext. 99% dos usuários trocam jpg→jpg.

### Decisão DEV-5 — Public bucket vs. private + signed URL

**Problema:** o bucket `avatars` pode ser `public: true` ou `public: false`.

**schema-design.md L1040** diz `public: false` ("URLs são signed"). Porém:

1. **Performance:** página pública (`/@<username>`) usa `next/image` com `priority` — signed URL exige round-trip extra antes de servir HTML inicial → LCP regredido.
2. **Simplicidade:** `getPublicUrl` é síncrono cliente-side; `createSignedUrl` é async + expira (precisa re-signar periodicamente).
3. **Conteúdo:** avatar é **público por design** (qualquer um pode ver `/@<username>` e o avatar dentro).
4. **RLS no objeto:** o `avatars_select_public` policy autoriza SELECT público mesmo com `public: false` no bucket — ambos os modos funcionam, mas `public: true` evita o round-trip de signing.

**Resolução:** **`public: true`** (override pragmático vs. schema-design.md L1040). Documentar em `0006_storage_avatars.sql` header + atualizar schema-design.md (append v0.5 — Task 1.7). **Pendente ratificação @data-engineer no handoff Task 1.**
